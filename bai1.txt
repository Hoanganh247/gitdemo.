<?php
function isPrime($n) {
    // Số nguyên tố phải lớn hơn 1
    if ($n < 2) {
        return false;
    }
    
    for ($i = 2; $i <= sqrt($n); $i++) {
        if ($n % $i == 0) {
            return false;
        }
    }
    
    return true;
}


echo "Danh sách các số nguyên tố từ 1 đến 100 là:\n";

for ($i = 1; $i <= 100; $i++) {
    if (isPrime($i)) {
        echo $i . " ";
    }
}
?>
