 <?php
function kiemTraSoNguyenTo($n) {
    if ($n < 2) {
        return false;
    }
    for ($i = 2; $i <= sqrt($n); $i++) { 
        if ($n % $i == 0) {
            return false;
        }
    }
    return true;
}
$num = 17;
if (kiemTraSoNguyenTo($num)) {
    echo "$num là số nguyên tố.";
} else {
    echo "$num không phải là số nguyên tố.";
}
?>
